p=/usr/local/bin/caddy
if [ -L "$p" ]; then
  echo type=link
  printf 'target=%s\n' "$(printf '%s' "$(readlink "$p")" | od -An -tx1 | tr -d ' \n')"
elif [ -d "$p" ]; then
  echo type=dir
  printf 'attrs=%s\n' "$(stat -c '%a %U %G' "$p" 2>/dev/null || stat -f '%Mp%Lp %Su %Sg' "$p")"
elif [ -f "$p" ]; then
  echo type=file
  printf 'attrs=%s\n' "$(stat -c '%a %U %G' "$p" 2>/dev/null || stat -f '%Mp%Lp %Su %Sg' "$p")"
  s=$(wc -c <"$p" | tr -d ' ')
  printf 'size=%s\n' "$s"
  if command -v sha256sum >/dev/null 2>&1; then h=$(sha256sum <"$p")
  elif command -v shasum >/dev/null 2>&1; then h=$(shasum -a 256 <"$p")
  else h=$(openssl dgst -sha256 -r <"$p"); fi
  h=${h%% *}
  printf 'sha256=%s\n' "$h"
elif [ -e "$p" ] || [ -h "$p" ]; then
  echo type=other
else
  echo type=absent
fi
